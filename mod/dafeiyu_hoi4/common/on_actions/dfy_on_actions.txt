on_actions = {
	on_startup = {
		effect = {
			every_country = {
				if = {
					limit = { is_ai = yes }
					dfy_log_state = yes
					dfy_log_relations = yes
				}
				if = {
					limit = { is_ai = no }
					dfy_log_state = yes
					dfy_log_relations = yes
				}
			}
		}
	}

	on_weekly = {
		effect = {
			if = {
				limit = { is_ai = yes }
				dfy_log_state = yes
				dfy_log_relations = yes
			}
			if = {
				limit = { is_ai = no }
				dfy_log_state = yes
				dfy_log_relations = yes
			}
		}
	}
}
